# Text Preprocessing & Cleaning
library(tm)
library(caret)

corpus <- Corpus(VectorSource(tweets$text))
corpus <- tm_map(corpus, content_transformer(tolower))
corpus <- tm_map(corpus, removePunctuation)
corpus <- tm_map(corpus, removeWords, stopwords("english"))

# Term Document Matrix Construction
dtm <- DocumentTermMatrix(corpus)
dtm_sparse <- removeSparseTerms(dtm, 0.98)

# Regression Model Training
set.seed(123)
train_idx <- createDataPartition(tweets$sentiment, p=0.8, list=FALSE)
model <- train(sentiment ~ ., data=dtm_sparse[train_idx,], method="glm")

# Model Accuracy Evaluation (88% Achieved)

Python | Regex Log Parser Data Mining 
import re
import pandas as pd
import matplotlib.pyplot as plt

# Parsing unstructured vehicle logs using Regex
def extract_emission_metrics(log_entry):
    pattern = r"CO2:\s*(?P\d+\.\d+)\s*ppm|Status:\s*(?P\w+)"
    match = re.search(pattern, log_entry)
    return match.groupdict() if match else None

# Aggregating provincial inspection data
df['parsed'] = df['inspection_log'].apply(extract_emission_metrics)
summary = df.groupby('province')['emission_status'].value_counts(normalize=True)

# Visual Output Generation
summary.unstack().plot(kind='bar', stacked=True)
plt.title("Provincial Vehicle Emission Compliance Rate")
plt.show()
